-- Generated TASK-044 verification from the same validated seed (review __REVIEW_HASH__).
\set ON_ERROR_STOP on
DO $verify$
DECLARE seed jsonb := __REVIEW_PAYLOAD__; actual jsonb; expected jsonb; role_name text; priv text; st record;
BEGIN
  -- Catalog and database agree row for row, including all 43 dispositioned ranges.
  SELECT jsonb_agg(to_jsonb(r) ORDER BY to_jsonb(r)::text) INTO actual FROM public.clinical_range_reviews r;
  SELECT jsonb_agg(r ORDER BY r::text) INTO expected FROM jsonb_array_elements(seed->'reviews') r;
  IF actual IS DISTINCT FROM expected THEN RAISE EXCEPTION 'review table differs from catalog seed'; END IF;
  IF (SELECT count(*) FROM public.clinical_range_reviews) <> 198
    OR (SELECT count(*) FROM public.clinical_range_reviews WHERE review_status='DISPUTED') <> 35
    OR (SELECT count(*) FROM public.clinical_range_reviews WHERE review_status='NEEDS_CONTEXT') <> 8
    OR (SELECT count(*) FROM public.clinical_range_reviews WHERE review_status='UNKNOWN') <> 155
    OR EXISTS (SELECT 1 FROM public.clinical_range_reviews WHERE review_status='SAFE')
    THEN RAISE EXCEPTION 'review disposition counts are not 198 = 35 DISPUTED + 8 NEEDS_CONTEXT + 155 UNKNOWN'; END IF;
  -- Exactly one review per frozen shipping range, and the frozen rows are untouched.
  IF EXISTS (SELECT 1 FROM public.clinical_ranges r WHERE r.candidate_id='shipping' AND (SELECT count(*) FROM public.clinical_range_reviews v
      WHERE v.slug=r.slug AND v.population=r.population AND v.phase IS NOT DISTINCT FROM r.phase AND v.kind=r.kind AND v.candidate_id=r.candidate_id) <> 1)
    OR EXISTS (SELECT 1 FROM public.clinical_range_reviews v WHERE NOT EXISTS (SELECT 1 FROM public.clinical_ranges r
      WHERE v.slug=r.slug AND v.population=r.population AND v.phase IS NOT DISTINCT FROM r.phase AND v.kind=r.kind AND v.candidate_id=r.candidate_id))
    THEN RAISE EXCEPTION 'review rows are not a bijection with V2-01 shipping ranges'; END IF;
  IF EXISTS (SELECT 1 FROM public.clinical_ranges WHERE candidate_id='shipping' AND review_status<>'UNKNOWN')
    THEN RAISE EXCEPTION 'V2-01 frozen review_status was edited'; END IF;
  SELECT * INTO STRICT st FROM clinical_archive.task044_state;
  IF st.status<>'applied' OR st.seed_hash<>'__REVIEW_HASH__' OR st.base_seed_hash<>'__BASE_HASH__' THEN RAISE EXCEPTION 'TASK-044 state mismatch'; END IF;
  IF NOT (SELECT relrowsecurity FROM pg_class WHERE oid='public.clinical_range_reviews'::regclass) THEN RAISE EXCEPTION 'review RLS disabled'; END IF;
  FOREACH role_name IN ARRAY ARRAY['anon','authenticated','service_role'] LOOP
    FOREACH priv IN ARRAY ARRAY['INSERT','UPDATE','DELETE','TRUNCATE','REFERENCES','TRIGGER'] LOOP
      IF has_table_privilege(role_name,'public.clinical_range_reviews',priv) THEN RAISE EXCEPTION '% has % on reviews',role_name,priv; END IF;
    END LOOP;
  END LOOP;
  IF has_table_privilege('anon','public.clinical_range_reviews','SELECT') THEN RAISE EXCEPTION 'anon can read reviews'; END IF;
  IF NOT has_table_privilege('authenticated','public.clinical_range_reviews','SELECT') THEN RAISE EXCEPTION 'authenticated cannot read reviews'; END IF;
  IF has_table_privilege('authenticated','clinical_archive.task044_state','SELECT') OR has_schema_privilege('authenticated','clinical_archive','USAGE')
    THEN RAISE EXCEPTION 'review state exposed'; END IF;
  -- SAFE stays impossible without a separately authorized evidence migration.
  BEGIN
    UPDATE public.clinical_range_reviews SET review_status='SAFE' WHERE slug='tsh' AND kind='amani_context' AND phase IS NULL;
    RAISE EXCEPTION 'review table accepted SAFE';
  EXCEPTION WHEN check_violation THEN NULL; END;
END $verify$;
